/*
 * ============================================================================
 * PAYMENT FLOW DIAGRAM
 * ============================================================================
 *
 *   [User Interface - Receptionist Billing Tab]
 *                    |
 *                    | 1. Click "Pay Bill"
 *                    ↓
 *   [Frontend - ReceptionistBillingTab.js]
 *                    |
 *                    | 2. POST /api/payments/create_order/
 *                    |    Body: { bill_id: 123 }
 *                    ↓
 *   [Backend - PaymentViewSet.create_order()]
 *                    |
 *                    | 3. Fetch bill details from MySQL
 *                    | 4. Create Razorpay order
 *                    | 5. Save to payment_transactions table
 *                    |
 *                    | Response: { order_id, amount, key }
 *                    ↓
 *   [Frontend - Razorpay Checkout Modal Opens]
 *                    |
 *                    | 6. User enters card details
 *                    | 7. Razorpay processes payment
 *                    |
 *                    | On Success
 *                    ↓
 *   [Frontend - Razorpay Handler Function]
 *                    |
 *                    | 8. POST /api/payments/verify_payment/
 *                    |    Body: {
 *                    |      razorpay_order_id,
 *                    |      razorpay_payment_id,
 *                    |      razorpay_signature
 *                    |    }
 *                    ↓
 *   [Backend - PaymentViewSet.verify_payment()]
 *                    |
 *                    | 9. Verify signature (HMAC-SHA256)
 *                    | 10. Update payment_transactions (status=captured)
 *                    | 11. Update billing (payment_status=Paid)
 *                    |
 *                    | Response: { status: success }
 *                    ↓
 *   [Frontend - Success Message + Refresh Data]
 *                    |
 *                    ✓ Payment Complete!
 *
 * ============================================================================
 */

/*
 * ============================================================================
 * DATABASE SCHEMA
 * ============================================================================
 *
 * Table: payment_transactions
 * ┌─────────────────────┬──────────────┬─────────────────────────────────┐
 * │ Column              │ Type         │ Description                     │
 * ├─────────────────────┼──────────────┼─────────────────────────────────┤
 * │ id                  │ INT (PK)     │ Auto-increment primary key      │
 * │ bill_id             │ INT (FK)     │ References billing.bill_id      │
 * │ razorpay_order_id   │ VARCHAR(100) │ Razorpay order ID (unique)      │
 * │ razorpay_payment_id │ VARCHAR(100) │ Payment ID (after success)      │
 * │ razorpay_signature  │ VARCHAR(255) │ Payment signature               │
 * │ amount              │ DECIMAL(10,2)│ Payment amount in INR           │
 * │ currency            │ VARCHAR(10)  │ Currency code (default: INR)    │
 * │ status              │ VARCHAR(20)  │ created/captured/failed         │
 * │ payment_method      │ VARCHAR(50)  │ card/upi/netbanking/wallet      │
 * │ error_code          │ VARCHAR(50)  │ Error code if failed            │
 * │ error_description   │ TEXT         │ Error description if failed     │
 * │ created_at          │ TIMESTAMP    │ Creation timestamp              │
 * │ updated_at          │ TIMESTAMP    │ Last update timestamp           │
 * └─────────────────────┴──────────────┴─────────────────────────────────┘
 *
 * Indexes:
 * - idx_bill_id (bill_id)
 * - idx_razorpay_order_id (razorpay_order_id)
 * - idx_status (status)
 *
 * Constraints:
 * - FOREIGN KEY (bill_id) REFERENCES billing(bill_id) ON DELETE CASCADE
 * - UNIQUE (razorpay_order_id)
 *
 * ============================================================================
 */

 *      "data": {
 *        "bill_id": 123,

 *    }

 *    GET /api/payments/transaction_history/?bill_id=123
 *    Auth: Required (JWT)
 *    Response: {
 *      "status": "success",
 *      "data": [ ... array of transactions ... ]
 *    }
 *
 * 4. WEBHOOK HANDLER
 *    POST /api/payments/handle_webhook/
 *    Auth: Not Required (Razorpay signature verified)
 *    Body: { ... Razorpay webhook payload ... }
 *
 * ============================================================================
 */

/*
 * ============================================================================
 * SIGNATURE VERIFICATION ALGORITHM
 * ============================================================================
 *
 * Python (Backend):
 * -----------------
 * import hmac
 * import hashlib
 * from django.conf import settings
 *
 * def verify_signature(order_id, payment_id, signature):
 *     message = f"{order_id}|{payment_id}"
 *     generated_signature = hmac.new(
 *         settings.RAZORPAY_SECRET_KEY.encode(),
 *         message.encode(),
 *         hashlib.sha256
 *     ).hexdigest()
 *     return generated_signature == signature
 *
 * Security: HMAC-SHA256 ensures payment authenticity
 *
 * ============================================================================
 */

/*
 * ============================================================================
 * PAYMENT STATUS LIFECYCLE
 * ============================================================================
 *
 *   created → authorized → captured → [END - Success]
 *      ↓
 *   failed → [END - Failure]
 *      ↓
 *   refunded → [END - Refund]
 *
 * Status Descriptions:
 * - created:    Order created, payment not initiated
 * - authorized: Payment authorized but not captured (for later capture)
 * - captured:   Payment successfully completed
 * - failed:     Payment failed
 * - refunded:   Payment refunded
 *
 * ============================================================================
 */
